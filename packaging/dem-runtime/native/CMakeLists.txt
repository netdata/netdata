cmake_minimum_required(VERSION 3.21)
project(netdata-dem-runtime LANGUAGES NONE)

foreach(required DEM_PAYLOAD_DIR DEM_ARCH DEM_VERSION DEM_RELEASE DEM_PACKAGE_FORMAT)
  if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
    message(FATAL_ERROR "${required} must be explicitly supplied")
  endif()
endforeach()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native packages must be assembled on Linux")
endif()
if(NOT DEM_ARCH MATCHES "^(amd64|arm64)$" OR NOT DEM_VERSION MATCHES "^[0-9][0-9A-Za-z.+~]*$"
   OR NOT DEM_RELEASE MATCHES "^[1-9][0-9]*$" OR NOT DEM_PACKAGE_FORMAT MATCHES "^(DEB|RPM)$")
  message(FATAL_ERROR "Unsupported architecture, package version/release or format")
endif()
if(NOT IS_ABSOLUTE "${DEM_PAYLOAD_DIR}" OR NOT EXISTS "${DEM_PAYLOAD_DIR}/native.json")
  message(FATAL_ERROR "DEM_PAYLOAD_DIR must be an absolute completed stage-native output")
endif()
file(READ "${DEM_PAYLOAD_DIR}/native.json" native_record)
string(JSON native_format GET "${native_record}" format)
string(JSON native_path GET "${native_record}" path)
string(JSON native_platform GET "${native_record}" manifest platform)
string(JSON native_version GET "${native_record}" manifest version)
if(NOT native_format EQUAL 1 OR NOT native_path STREQUAL "/usr/lib/netdata/dem-runtime"
   OR NOT native_platform STREQUAL "linux/${DEM_ARCH}" OR NOT native_version STREQUAL DEM_VERSION)
  message(FATAL_ERROR "Staged native metadata differs from requested package identity or fixed prefix")
endif()

# CMake-created ancestors must not inherit the builder's umask. Payload modes
# remain those supplied by stage-native through USE_SOURCE_PERMISSIONS.
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(DIRECTORY "${DEM_PAYLOAD_DIR}/" DESTINATION usr/lib/netdata/dem-runtime USE_SOURCE_PERMISSIONS)
install(FILES README.md DESTINATION usr/share/doc/netdata-dem-runtime)
# Conventional boot-loaded policy, preserving administrator edits on upgrades.
install(FILES netdata-dem-browser DESTINATION etc/apparmor.d
        PERMISSIONS OWNER_READ OWNER_WRITE GROUP_READ WORLD_READ)
set(CPACK_RPM_USER_FILELIST "%config(noreplace) /etc/apparmor.d/netdata-dem-browser")
list(APPEND CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION "/etc/apparmor.d")
set(CPACK_GENERATOR "${DEM_PACKAGE_FORMAT}")
set(CPACK_PACKAGE_NAME netdata-dem-runtime)
set(CPACK_PACKAGE_VERSION "${DEM_VERSION}")
set(CPACK_PACKAGE_VENDOR "Netdata Inc.")
set(CPACK_PACKAGE_CONTACT "Netdata Builder <bot@netdata.cloud>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Optional private browser runtime for Netdata DEM")
set(CPACK_PACKAGE_DESCRIPTION "Independent browser runtime with package-owned payload and coordinated native admission. See included component notices and source inventory.")
set(CPACK_PACKAGING_INSTALL_PREFIX "/")
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}/packages")
set(CPACK_STRIP_FILES OFF)
# The payload is bound to one final prefix; CPack relocation also rewrites
# relative symlinks into absolute paths that the confined reader must reject.
set(CPACK_PACKAGE_RELOCATABLE FALSE)
set(CPACK_RPM_PACKAGE_RELOCATABLE OFF)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS OFF)
set(CPACK_DEBIAN_PACKAGE_GENERATE_SHLIBS OFF)
set(CPACK_DEBIAN_DEBUGINFO_PACKAGE OFF)
set(CPACK_RPM_DEBUGINFO_PACKAGE OFF)
set(CPACK_RPM_PACKAGE_AUTOREQPROV OFF)
set(CPACK_RPM_PACKAGE_AUTOREQ OFF)
set(CPACK_RPM_PACKAGE_AUTOPROV OFF)
set(CPACK_RPM_DEFAULT_USER root)
set(CPACK_RPM_DEFAULT_GROUP root)
# RPM's install post-processing can rewrite ELF files after fixed-slot binding.
# CPack has no dedicated Requires(posttrans) variable; insert the header explicitly.
set(CPACK_RPM_SPEC_MORE_DEFINE "%define __spec_install_post /bin/true\nRequires(posttrans): netdata-dem-runtime-api = 1, systemd, coreutils, sed")
list(APPEND CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION "/usr/lib/netdata")

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/lifecycle.sh" NATIVE_LIFECYCLE)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/hooks")
if(DEM_PACKAGE_FORMAT STREQUAL "DEB")
  set(NATIVE_FAMILY deb)
  foreach(hook preinst postinst prerm postrm)
    set(NATIVE_HOOK "deb-${hook}")
    configure_file(hook.sh.in "hooks/${hook}" @ONLY)
    file(CHMOD "${CMAKE_BINARY_DIR}/hooks/${hook}" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
    list(APPEND CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_BINARY_DIR}/hooks/${hook}")
  endforeach()
  file(WRITE "${CMAKE_BINARY_DIR}/hooks/conffiles" "/etc/apparmor.d/netdata-dem-browser\n")
  list(APPEND CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_BINARY_DIR}/hooks/conffiles")
  set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
  set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "${DEM_ARCH}")
  set(CPACK_DEBIAN_PACKAGE_RELEASE "${DEM_RELEASE}")
  set(CPACK_DEBIAN_PACKAGE_SECTION net)
  set(CPACK_DEBIAN_PACKAGE_PREDEPENDS "netdata-dem-runtime-api (= 1), init-system-helpers, systemd, coreutils, sed")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
else()
  set(NATIVE_FAMILY rpm)
  foreach(hook pre preun posttrans postun)
    set(NATIVE_HOOK "rpm-${hook}")
    configure_file(hook.sh.in "hooks/${hook}" @ONLY)
  endforeach()
  set(CPACK_RPM_PRE_INSTALL_SCRIPT_FILE "${CMAKE_BINARY_DIR}/hooks/pre")
  set(CPACK_RPM_PRE_UNINSTALL_SCRIPT_FILE "${CMAKE_BINARY_DIR}/hooks/preun")
  set(CPACK_RPM_POST_TRANS_SCRIPT_FILE "${CMAKE_BINARY_DIR}/hooks/posttrans")
  set(CPACK_RPM_POST_UNINSTALL_SCRIPT_FILE "${CMAKE_BINARY_DIR}/hooks/postun")
  if(DEM_ARCH STREQUAL "amd64")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE x86_64)
  else()
    set(CPACK_RPM_PACKAGE_ARCHITECTURE aarch64)
  endif()
  set(CPACK_RPM_PACKAGE_RELEASE "${DEM_RELEASE}")
  set(CPACK_RPM_PACKAGE_LICENSE "Multiple (see bundled component notices)")
  set(CPACK_RPM_PACKAGE_REQUIRES "netdata-dem-runtime-api = 1, systemd, coreutils, sed")
  set(CPACK_RPM_PACKAGE_REQUIRES_PRE "netdata-dem-runtime-api = 1, systemd, coreutils, sed")
  set(CPACK_RPM_PACKAGE_REQUIRES_PREUN "netdata-dem-runtime-api = 1, systemd, coreutils, sed")
  set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
endif()
include(CPack)
